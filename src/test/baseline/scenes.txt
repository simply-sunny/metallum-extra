# Phase 0 baseline: fixed scenes for comparing renderers, run by run.sh. Every scene is shot (with a .txt of what it was
# taken with) and then timed for 10 seconds. Wind and waves are stood still by -Dmetallumextra.freezeTime=true.
world PackTest
wait 2000
screen close
hud off
cmd gamerule advance_time false
cmd gamerule advance_weather false
cmd weather clear
cmd time set 2000
cmd difficulty peaceful
pack builtin
set shaders on
set quality medium
set shadows on
set bloom on
set reflections on
set waving on
set rays off
set ao on
set colored on
set edges on
cmd gamemode spectator @a
cmd tp @a 0.5 203 -9 0 25
wait 3000
# The arena: a grass platform with stone, leaves, grass, a pool, lights and mobs.
cmd fill -12 199 -12 12 199 12 minecraft:grass_block
cmd fill -12 200 -12 12 215 12 minecraft:air
cmd fill 3 200 3 3 210 3 minecraft:stone
cmd fill -4 200 5 -2 203 7 minecraft:stone
cmd fill -3 203 2 3 205 6 minecraft:oak_leaves
cmd fill -6 200 0 6 200 8 minecraft:short_grass
cmd fill -6 200 -3 6 201 -3 minecraft:tall_grass
cmd fill 6 198 -11 11 198 -5 minecraft:stone
cmd fill 6 199 -11 11 199 -5 minecraft:water
cmd setblock -8 200 -6 minecraft:torch
cmd setblock -9 200 -8 minecraft:soul_lantern
cmd setblock -10 200 -6 minecraft:glowstone
cmd setblock -7 200 -9 minecraft:lantern
cmd summon minecraft:cow -9 200 2 {NoAI:1b,NoGravity:1b,Rotation:[20f,0f]}
cmd summon minecraft:pig -7 200 3 {NoAI:1b,NoGravity:1b,Rotation:[-30f,0f]}
cmd summon minecraft:armor_stand -5 200 2 {NoGravity:1b,ShowArms:1b,Pose:{RightArm:[-40f,20f,0f]}}
cmd summon minecraft:item -11 200 3 {Item:{id:"minecraft:diamond",count:1},NoGravity:1b,Age:-32768s}
cmd summon minecraft:item_frame -12 201 6 {Facing:5b,Item:{id:"minecraft:compass",count:1}}
wait 4000
